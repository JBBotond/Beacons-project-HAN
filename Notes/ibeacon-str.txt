OK+DISC + factoryID = 16bytes
iBeaconUUID = 32bytes
MajorValue+MeasuredPower = 10bytes
MAC = 12bytes
RSSI = 4bytes

Memory required = 74+1bytes(excluding ':', \n and \r characters)
